```cpp
vector<int> add(const vector<int>& a, const vector<int>& b) {
    vector<int> c;
    long long t = 0;
    for (size_t i = 0; i < max(a.size(), b.size()) || t; ++i) {
```

```cpp
vector<int> sub(const vector<int>& a, const vector<int>& b) {
    vector<int> c = a;
    for (size_t i = 0; i < c.size(); ++i) {
        if (i < b.size()) c[i] -= b[i];
        if (c[i] < 0) {
            c[i] += BASE;
            c[i + 1]--;
        }
    }
    while (c.size() > 1 && c.back() == 0) c.pop_back(); // 去除前导 0
    return c;
}
```

```cpp
vector<int> mul(const vector<int>& a, const vector<int>& b) {
    vector<int> c(a.size() + b.size(), 0);
    for (size_t i = 0; i < a.size(); ++i) {
        long long t = 0;
        for (size_t j = 0; j < b.size() || t; ++j) {
            long long cur = c[i + j] + t + 1LL * a[i] * (j < b.size() ? b[j] : 0);
            c[i + j] = cur % BASE;
            t = cur / BASE;
        }
    }
    while (c.size() > 1 && c.back() == 0) c.pop_back();
    return c;
}
```

```cpp
vector<int> div(const vector<int>& a, long long b, long long &r) {
    vector<int> c(a.size());

vector<int> mul(const vector<int>& a, long long b) {
    if (b == 0 || (a.size() == 1 && a[0] == 0)) return {0};
    vector<int> c;
    long long t = 0;
    for (size_t i = 0; i < a.size() || t; ++i) {
        if (i < a.size()) t += 1LL * a[i] * b;
        c.push_back(t % BASE);
        t /= BASE;
    }
    while (c.size() > 1 && c.back() == 0) c.pop_back();
    return c;
}

// 高精 / 高精: 返回 {商 Q, 余数 R}
pair<vector<int>, vector<int>> div_big(const vector<int>& a, const vector<int>& b) {
    if (!cmp(a, b)) return {{0}, a}; // a < b 时，商 0，余数为 a
    
    vector<int> q(a.size(), 0);
    vector<int> r = {0};

    // 从最高位块开始往下处理
    for (int i = (int)a.size() - 1; i >= 0; --i) {
        // 余数乘 BASE 并加上 a[i] (即在低端插入 a[i])
        if (r.size() == 1 && r[0] == 0) r.clear();
        r.insert(r.begin(), a[i]);
        while (r.size() > 1 && r.back() == 0) r.pop_back();

        // 二分确定当前压位块的商 (0 ~ BASE - 1)
        int low = 0, high = BASE - 1, digit = 0;
        while (low <= high) {
            int mid = low + (high - low) / 2;
            if (cmp(r, mul(b, mid))) {
                digit = mid;
                low = mid + 1;
            } else {
                high = mid - 1;
            }
        }
        q[i] = digit;
        r = sub(r, mul(b, digit));
    }
    while (q.size() > 1 && q.back() == 0) q.pop_back();
    return {q, r};
}
```

```cpp
void exgcd(ll a,ll b,ll& x,ll& y){
    // 引用传值x,y
    if(b==0){
        x = 1,y = 0;
        return;
    }

    /*
        初始的x,y只是占位的用处，到递归最深处会有真正的值
        这个时候返回x,y
        然后我们推出，此时返回的x，y，与这一层的a，b关系是
        ax + b[ y - (a/b)*x ] = (a,b)
        于是我们需要调整一下y，让a,b,x,y满足 ax + by = (a,b)
        故 y -= (a/b) * x;
    */

    // 这里有个小trick，y和x反写
    exgcd(b, a%b ,y,x);
    
    y -= (a/b) * x;
}
```

3. 高斯消元
```cpp
const double EPS = 1e-9; // 精度阈值

/*
 * 高斯-约当消元（Gauss-Jordan Elimination）
 * 参数：
 *   a : 增广矩阵，大小为 n 行 × (m+1) 列，其中 n 为方程个数，m 为未知数个数。
 *       （通常 n = m，即方阵）
 *   x : 用于存储解向量（当有唯一解时）
 * 返回值：
 *   0 : 唯一解
 *   1 : 无穷多解
 *   2 : 无解
 */
int gauss(vector<vector<double>> a, vector<double> &x)
{
    int n = (int)a.size();        // 方程数
    int m = (int)a[0].size() - 1; // 未知数个数

    vector<int> where(m, -1); // where[col] = 行号，表示变量 col 对应的主元所在行（-1表示自由变量）

    int row = 0; // 当前处理的行
    // 按列扫描
    for (int col = 0; col < m && row < n; ++col)
    {
        // 1. 选主元：在 row 到 n-1 行中，找到当前列绝对值最大的行
        int sel = row;
        for (int i = row; i < n; ++i)
        {
            if (fabs(a[i][col]) > fabs(a[sel][col]))
            {
                sel = i;
            }
        }
        // 如果该列所有元素都接近 0，则此列为自由变量，跳过
        if (fabs(a[sel][col]) < EPS)
            continue;
        // 2. 将主元行交换到当前行
        swap(a[sel], a[row]);
        // 记录该列对应的主元行
        where[col] = row;
        // 3. 归一化主元行：将主元变为 1
        double pivot = a[row][col];
        for (int j = col; j <= m; ++j)
        {
            a[row][j] /= pivot;
        }
        // 4. 消去其他所有行（包括上方和下方），使该列除主元外均为 0
        for (int i = 0; i < n; ++i)
        {
            if (i == row)
                continue;
            double factor = a[i][col];
            for (int j = col; j <= m; ++j)
            {
                a[i][j] -= factor * a[row][j];
            }
        }
        ++row; // 处理下一行
    }
    // 5. 检查无解：若存在形如 0 = b (b ≠ 0) 的行，则无解
    for (int i = row; i < n; ++i)
    {
        if (fabs(a[i][m]) > EPS)
        {
            return 2; // 无解
        }
    }
    // 6. 判断是否有自由变量（即存在某些列没有主元）
    if (row < m)
    {
        return 1; // 无穷多解
    }
    // 7. 唯一解：直接读取每个主元列对应的常数项
    x.assign(m, 0);
    for (int col = 0; col < m; ++col)
    {
        // where[col] 是主元所在行，该行的最后一列即为解
        x[col] = a[where[col]][m];
    }
    return 0;
}

// ------------------------------------------------------------
// 以下是使用示例
int main()
{
    // 示例：求解方程组
    // 2x +  y -  z = 8
    // -3x -  y + 2z = -11
    // -2x +  y + 2z = -3
    vector<vector<double>> a = {
        {2, 1, -1, 8},
        {-3, -1, 2, -11},
        {-2, 1, 2, -3}};

    vector<double> x;
    int res = gauss(a, x);

    if (res == 0)
    {
        cout << "唯一解：\n";
        for (int i = 0; i < (int)x.size(); ++i)
        {
            cout << "x" << i << " = " << x[i] << "\n";
        }
    }
    else if (res == 1)
    {
        cout << "无穷多解\n";
    }
    else
    {
        cout << "无解\n";
    }

    return 0;
}
```

```cpp
int p;

ll qmi(ll a,ll p){
    // 依旧快速幂求 模质数的逆元
    ll b = p-2;
    ll t = a;
    ll res = 1;
    while(b){
        if(b&1) res = res * t % p;
        b >>= 1;
        t = t * t % p; 
    }
    return res;
}


ll C(int a,int b){
    // 这里保证 a,b < 1e5，直接朴素就可以
    if(b==0 || a == b) return 1;
    if(a<b) return 0;
    ll res = 1,tmp = 1;
    for(int i = a,j = b;j>0;i--,j--){
        res = res * i % p;
        tmp = tmp * j % p;
    }
    res = res * qmi(tmp,p) % p;
    return res;
}


ll lucas(ll a,ll b){
    // Lucas 定理 Ca,b mod p = Ca%p,b%p * Ca/p,b/p mod p;
    if(a < p && b < p) return C(a,b) % p;
    else{
        return C(a%p,b%p) * lucas(a/p,b/p) % p;
    }
}
```

```cpp
    int n;

    vector<int> st(n+1),euler(n+1),prime;
    /*
    
    st记录这个数是否被标记过，prime记录是否是质数，euler记录欧拉函数值
    关键还是在那个欧拉函数的计算公式，然后递推
    但是要注意一个点，可以往后推的前提是当前的算过。那是自然，当前的已经被前面筛过了。
    如果没筛过说明是素数，那么循环前已处理
    */

        if(!st[i]){
            euler[i] = i-1;
            prime.push_back(i);
        }
        for(int j = 0;prime[j]<=n/i;j++){
            int p = prime[j];
            st[i * p] = 1;
            
            // 下面递推公式要搞清楚
            if(i % p != 0)
                euler[i*p] = (p-1)*euler[i];
            else{
                euler[i*p] = p*euler[i];
                break;
            }
        }
    }

    long long res = 0;
    for(int i = 1;i<=n;i++){
        res += euler[i];
    }
    cout << res;
```

2. 线段树
```cpp
struct node{
    ll l,r;
    ll v,lazy;
};

const int N = 1e5+10;
ll arr[N];
node tr[4*N];

void pushup(ll u){
    tr[u].v = tr[u<<1].v + tr[u<<1|1].v;
    tr[u].l = tr[u<<1].l;
    tr[u].r = tr[u<<1|1].r;
}
void build(ll u,ll l,ll r){
    if(l>r) return; 
    if(l==r){
        tr[u] = {l,r,arr[l],0};
        return;
    }
    ll mid = l + (r-l)/2;
    build(2*u,l,mid),build(2*u+1,mid+1,r);
    pushup(u);
}

void pushdown(ll u){
    if(tr[u].l == tr[u].r) return;
    if(tr[u].lazy==0) return;
    tr[u<<1].v += (tr[u<<1].r - tr[u<<1].l + 1) * tr[u].lazy;
    tr[u<<1|1].v += (tr[u<<1|1].r - tr[u<<1|1].l + 1) * tr[u].lazy;
    tr[u << 1].lazy += tr[u].lazy, tr[u << 1 | 1].lazy += tr[u].lazy;
    tr[u].lazy = 0;
}

void qmodify(ll aiml,ll aimr,ll u,ll lazy){
    ll ql = tr[u].l,qr = tr[u].r;
    if(aiml>qr || aimr < ql) return;
    if(aiml<=ql && aimr >= qr){
        tr[u].v += (qr-ql+1) * lazy;
        tr[u].lazy += lazy;
        return;
    }
    ll mid = ql + (qr-ql)/2;
    pushdown(u);
    if(aiml <= mid) qmodify(aiml,aimr,u*2,lazy);
    if(aimr >= mid+1) qmodify(aiml,aimr,u<<1|1,lazy);
    pushup(u);
}

ll query(ll aiml,ll aimr,ll u){
    ll ql = tr[u].l,qr = tr[u].r;
    if(aiml <= ql && aimr >= qr) return tr[u].v;
    ll res = 0;
    ll mid = ql + (qr-ql)/2;
    pushdown(u);
    if(aiml<=mid) res += query(aiml,aimr,u<<1);
    if(aimr >= mid + 1) res += query(aiml,aimr,u<<1|1);
    return res;
}
```

2. AC 自动机
```cpp
#include <bits/stdc++.h>
using namespace std;

const int N = 10010;        // 模式串总长度（即 Trie 树节点数上限，根据题目调整）
const int M = 1000010;      // 文本串长度上限（根据题目调整）
const int SIGMA = 26;       // 字符集大小（小写字母）

int tr[N][SIGMA];   // Trie 树的子节点指针，tr[u][c] 表示节点 u 经过字符 c 到达的子节点编号（0 表示不存在）
int fail[N];        // fail 指针：失配时跳转到的节点
int cnt[N];         // cnt[u] 表示以节点 u 结尾的模式串数量（可用于统计出现次数）
int idx;            // 节点编号分配器，0 号节点为根节点

/**
 * 插入一个模式串
 * s : 模式串（小写字母）
 */
void insert(char s[]) {
    int u = 0;
    for (int i = 0; s[i]; i++) {
        int c = s[i] - 'a';
        if (!tr[u][c]) tr[u][c] = ++idx;   // 新建节点
        u = tr[u][c];
    }
    cnt[u]++;   // 该节点对应的模式串数量 +1（若存在重复串，则统计多次）
}

/**
 * 构建 AC 自动机的 fail 指针（BFS）
 */
void build() {
    queue<int> q;
    // 第一层（深度为 1）的节点，fail 指针直接指向根节点 0
    for (int c = 0; c < SIGMA; c++) {
        if (tr[0][c]) {
            fail[tr[0][c]] = 0;
            q.push(tr[0][c]);
        }
    }

    // BFS 逐层构建
    while (!q.empty()) {
        int u = q.front(); q.pop();
        for (int c = 0; c < SIGMA; c++) {
            int v = tr[u][c];
            if (v) {
                // v 存在：其 fail 指针指向 u 的 fail 指针指向的节点的 c 子节点
                fail[v] = tr[fail[u]][c];
                q.push(v);
            } else {
                // 优化：不存在的边直接指向 fail 指针的对应转移，减少跳转
                tr[u][c] = tr[fail[u]][c];
            }
        }
    }
}

/**
 * 查询文本串中所有模式串出现的总次数
 * s : 文本串
 * 返回值：匹配到的模式串总次数（每个模式串重复出现也累加）
 */
int query(char s[]) {
    int u = 0, res = 0;
    for (int i = 0; s[i]; i++) {
        int c = s[i] - 'a';
        u = tr[u][c];   // 利用优化后的转移边直接跳转
        // 遍历所有可能的匹配（通过 fail 链）
        for (int j = u; j && cnt[j] != -1; j = fail[j]) {
            res += cnt[j];
            cnt[j] = -1;   // 标记已经统计过，防止重复计数（若需要统计每个模式串出现次数，则不能置 -1）
        }
    }
    return res;
}

int main() {
    int T;
    scanf("%d", &T);
    while (T--) {
        // 初始化
        memset(tr, 0, sizeof tr);
        memset(fail, 0, sizeof fail);
        memset(cnt, 0, sizeof cnt);
        idx = 0;
```

```cpp
#include <bits/stdc++.h>
using namespace std;

typedef unsigned long long ULL; // 自然溢出（懒人专用），但容易被卡，这里采用双模数
typedef pair<int, int> PII;

// ============ 双哈希参数（两套进制 + 两个大质数）============
const int N = 100010;          // 根据字符串最大长度调整
const int P1 = 131, P2 = 13331;
const int MOD1 = 1e9 + 7, MOD2 = 1e9 + 9;

int n;                         // 字符串长度
char str[N];                   // 原字符串（下标从1开始存）
int h1[N], h2[N];              // 前缀哈希值
int p1[N], p2[N];              // 进制幂次

// 初始化哈希（传入字符串，注意下标从1开始）
void init_hash(char s[]) {
    p1[0] = p2[0] = 1;
    for (int i = 1; s[i]; i ++ ) {
        int v = s[i] - 'a' + 1;      // 映射为1-26，避免前导零
        
        p1[i] = 1LL * p1[i-1] * P1 % MOD1;
        p2[i] = 1LL * p2[i-1] * P2 % MOD2;
        
        h1[i] = (1LL * h1[i-1] * P1 + v) % MOD1;
        h2[i] = (1LL * h2[i-1] * P2 + v) % MOD2;
    }
    n = strlen(s + 1); // 记录长度
}

// 获取子串 s[l..r] 的双哈希值 (l, r 从1开始)
PII get_hash(int l, int r) {
    int len = r - l + 1;
    int res1 = (h1[r] - 1LL * h1[l-1] * p1[len] % MOD1 + MOD1) % MOD1;
    int res2 = (h2[r] - 1LL * h2[l-1] * p2[len] % MOD2 + MOD2) % MOD2;
    return {res1, res2};
}

// ==================== main 调用示例 ====================
int main() {
    // 1. 输入字符串，下标从1开始存放
    scanf("%s", str + 1); 
    
    // 2. 预处理哈希
    init_hash(str);
    
    // 3. 查询示例：比较 [1,3] 和 [4,6] 是否相等
    PII a = get_hash(1, 3);
    PII b = get_hash(4, 6);
    if (a == b) puts("相等");
    else puts("不相等");
    
    // 4. 配合 map 进行计数判重
    map<PII, int> cnt;
    cnt[get_hash(1, 2)] ++;  // 子串1-2
    cnt[get_hash(3, 4)] ++;  // 子串3-4
    cnt[get_hash(1, 2)] ++;  // 再次出现
    
    printf("子串[1,2]出现次数: %d\n", cnt[get_hash(1, 2)]);
    
    return 0;

    }
```

3. SPFA
```cpp
#include <bits/stdc++.h>
using namespace std;

const int N = 100010;          // 图中点的最大数量（根据题目调整）
const int INF = 0x3f3f3f3f;    // 定义一个很大的数，表示“无穷大”

// 邻接表边的结构体
struct datas {
    int node;   // 边的终点
    int w;      // 边的权值
};

vector<datas> g[N];   // g[u] 存储所有从 u 出发的边
int dist[N];          // dist[i] 表示起点到 i 的最短距离
bool st[N];           // st[i] = true 表示点 i 当前在队列中（等待被松弛）

/**
 * SPFA 算法（队列优化的 Bellman-Ford）
 * 参数 s：源点（起点）
 */
void spfa(int s) {
    // 1. 初始化距离数组为无穷大
    memset(dist, 0x3f, sizeof dist);
    dist[s] = 0;          // 起点到自己的距离为 0

    // 2. 创建队列，将起点入队
    queue<int> q;
    q.push(s);
    st[s] = true;         // 标记起点已经在队列中

    // 3. 不断从队列中取出点进行松弛操作
    while (!q.empty()) {
        int u = q.front(); q.pop();   // 取出队首元素 u
        st[u] = false;                // ★★★ 关键点：u 已经离开队列，取消标记

        /*
         * 解释 st[u] = false 的作用：
         * st 数组用来记录“当前节点是否已经在队列中”。
         * 当我们将 u 从队列中取出时，它暂时不在队列里了，
         * 所以要把标记改为 false，这样如果后续 u 的距离再次被更新，
         * 它就可以重新入队（因为 if (!st[v]) 条件成立）。
         * 
         * 如果这里不置 false，那么即使 u 的距离被更新了，
         * 由于 st[u] 仍为 true，它也不会再次入队，
         * 导致后续可能更优的路径无法被处理，算法就会出错。
         * 
         * 简而言之：st 标记的是“是否在队列中”，出队就要清除标记。
         */

        // 4. 遍历所有从 u 出发的边 (u -> v)
        for (auto e : g[u]) {
            int v = e.node;      // 邻接点
            int w = e.w;         // 边权

            // 5. 如果通过 u 可以使得到 v 的距离变短，就更新
            if (dist[v] > dist[u] + w) {
                dist[v] = dist[u] + w;

                // 6. 如果 v 当前不在队列中，就将其入队（避免重复入队）
                if (!st[v]) {
                    q.push(v);
                    st[v] = true;   // 标记 v 已入队
                }
            }
        }
    }
    // 当队列为空时，所有能更新的距离都处理完了，算法结束
}
```

```cpp
const int N = 210;          // 最大点数，根据题目调整（通常 100~500）

/**
 * Floyd 算法（动态规划，多源最短路）
 * 参数 n：图中点的数量（编号 1 ~ n）
 * 
 * 算法核心：三重循环，枚举中间点 k，尝试用 i->k->j 更新 i->j
 */
void floyd(int n) {
    // 三重循环，k 必须放在最外层
    for (int k = 1; k <= n; k++) {
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n; j++) {
                // 如果通过 k 能使得距离更短，则更新
                if (dist[i][j] > dist[i][k] + dist[k][j]) {
                    dist[i][j] = dist[i][k] + dist[k][j];
                }
            }
        }
    }
}
```

### 最小生成树
1. 朴素版prim
```cpp
const int N = 510;          // 最大点数，根据题目调整（一般 500 以内）
const int INF = 0x3f3f3f3f; // 表示无穷大

int g[N][N];    // 邻接矩阵，g[i][j] 表示边 i->j 的权值（无向图则对称）
int dist[N];    // dist[i] 表示点 i 到当前已选生成树集合的最小距离
bool st[N];     // st[i] = true 表示点 i 已经在生成树中

/**
 * 朴素 Prim 算法求最小生成树
 * 参数 n：图中点的数量（编号从 1 到 n）
 * 返回值：最小生成树的边权之和，若图不连通则返回 INF
 */
int prim(int n) {
    // 1. 初始化距离数组为无穷大
    memset(dist, 0x3f, sizeof dist);
    // 2. 所有点初始都未加入生成树
    memset(st, false, sizeof st);

    // 3. 通常选择 1 号点作为起点，将其到集合的距离设为 0
    dist[1] = 0;

    int res = 0;   // 记录最小生成树的总权值

    // 4. 迭代 n 次，每次将一个点加入生成树
    for (int i = 0; i < n; i++) {
        // 4.1 在未加入生成树的点中，找到距离集合最近的点 t
        int t = -1;
        for (int j = 1; j <= n; j++) {
            if (!st[j] && (t == -1 || dist[t] > dist[j])) {
                t = j;
            }
        }

        // 4.2 如果 t 的距离仍为 INF（且不是第一次迭代），说明图不连通
        //     （因为第一次迭代时 dist[1]=0，一定能找到点）
        if (dist[t] == INF) return INF;

        // 4.3 将 t 加入生成树，累加边权
        res += dist[t];
        st[t] = true;

        // 4.4 用新加入的点 t 更新其他未加入点到生成树的距离
        for (int j = 1; j <= n; j++) {
            if (!st[j] && dist[j] > g[t][j]) {
                dist[j] = g[t][j];
            }
        }
    }

    return res;
}
```

2. kruskal
```cpp
// 边结构体：存储一条无向边（a, b, w）
struct Edge {
    int a, b, w;
    // 重载 < 运算符，用于按权值升序排序
    bool operator< (const Edge &E) const {
        return w < E.w;
    }
} edges[M];

int p[N];   // 并查集父数组，p[x] 表示 x 的父节点

/**
 * 并查集查找函数（路径压缩）
 * 返回 x 所在集合的根节点
 */
int find(int x) {
    if (p[x] != x) p[x] = find(p[x]);   // 递归路径压缩
    return p[x];
}

/**
 * Kruskal 算法求最小生成树
 * 参数 n：图中点的数量（编号从 1 到 n）
 * 参数 m：边的数量（使用全局 edges[0..m-1]）
 * 返回值：最小生成树的总权值，若图不连通则返回 INF
 */
int kruskal(int n, int m) {
    // 1. 将所有边按权值从小到大排序
    sort(edges, edges + m);

    // 2. 初始化并查集：每个点自成一个集合
    for (int i = 1; i <= n; i++) p[i] = i;

    int res = 0;   // 记录最小生成树的总权值
    int cnt = 0;   // 记录当前生成树中已添加的边数

    // 3. 从小到大遍历每条边
    for (int i = 0; i < m; i++) {
        int a = edges[i].a, b = edges[i].b, w = edges[i].w;

        // 3.1 查找 a 和 b 所属的集合根节点
        int pa = find(a), pb = find(b);

        // 3.2 如果 a 和 b 不在同一个集合中，则加入这条边
        if (pa != pb) {
            p[pa] = pb;          // 合并两个集合
            res += w;            // 累加边权
            cnt++;               // 边数加 1
        }
        // 若在同一个集合，则跳过（否则会形成环）
    }

    // 4. 判断是否连通：如果生成树包含的边数不足 n-1，说明图不连通
    if (cnt < n - 1) return INF;

    return res;
}
```

### 二分图
1. 二染色
```cpp
vector<int> g[N];       // 邻接表，g[u] 存储所有与 u 相邻的点
int color[N];           // 染色数组：0 表示未染色，1 和 2 表示两种不同颜色

/**
 * DFS 染色函数（递归实现）
 * 参数 u：当前点编号
 * 参数 c：要给 u 染的颜色（1 或 2）
 * 返回值：若染色成功返回 true，发生冲突返回 false
 */
bool dfs(int u, int c) {
    color[u] = c;   // 给 u 染上颜色 c

    // 遍历所有邻接点 v
    for (int v : g[u]) {
        if (!color[v]) {
            // 如果 v 未染色，则染成相反颜色（3 - c），并递归检查
            if (!dfs(v, 3 - c)) return false;
        } else if (color[v] == c) {
            // 如果 v 已染色且与 u 同色，说明出现冲突，不是二分图
            return false;
        }
    }
    return true;
}

/**
 * 二分图判定函数
 * 参数 n：图中点的数量（编号从 1 到 n）
 * 返回值：是二分图返回 true，否则返回 false
 */
bool isBipartite(int n) {
    memset(color, 0, sizeof color);   // 初始化所有点未染色

    // 遍历每个点（图可能不连通）
    for (int i = 1; i <= n; i++) {
        if (!color[i]) {
            // 如果 i 未染色，从 i 开始用颜色 1 进行 DFS 染色
            if (!dfs(i, 1)) return false;
        }
    }
    return true;
}
```

2. 匈牙利算法
```cpp
vector<int> g[N];       // 邻接表，g[u] 存储左部点 u 能匹配的右部点
int match[N];           // match[v] 表示右部点 v 当前匹配的左部点（0 表示未匹配）
bool st[N];             // st[v] 在单次寻找增广路时标记右部点是否被访问过

/**
 * 为左部点 u 寻找增广路（DFS）
 * 参数 u：左部点编号
 * 返回值：若找到增广路则 true，否则 false
 */
bool find(int u) {
    // 遍历 u 所有可匹配的右部点 v
    for (int v : g[u]) {
        if (!st[v]) {
            st[v] = true;   // 标记 v 本次已访问，避免递归中重复

            // 如果 v 未匹配，或 v 已匹配但能为其当前匹配的左部点找到其他右部点
            if (match[v] == 0 || find(match[v])) {
                match[v] = u;   // 将 v 匹配给 u
                return true;
            }
        }
    }
    return false;
}

/**
 * 匈牙利算法求最大匹配数
 * 参数 n1：左部点数量（编号 1 ~ n1）
 * 参数 n2：右部点数量（编号 1 ~ n2，但通常用 match 下标即可）
 * 返回值：最大匹配数
 */
int hungarian(int n1, int n2) {
    memset(match, 0, sizeof match);   // 初始化所有右部点未匹配

    int res = 0;
    for (int i = 1; i <= n1; i++) {
        memset(st, false, sizeof st); // 每次尝试前清空访问标记
        if (find(i)) res++;           // 如果找到增广路，匹配数 +1
    }
    return res;
}
```

```cpp
    for (int i = 1; i <= n; i ++ )
        for (int j = 1; j <= m; j ++ )
            s[i][j] = s[i - 1][j] + s[i][j - 1] - s[i - 1][j - 1] + a[i][j];
}

// 查询子矩阵 (x1,y1) 到 (x2,y2) 的和（坐标均从1开始）
// 时间复杂度 O(1)
int query_2d_prefix(int x1, int y1, int x2, int y2) {
    return s[x2][y2] - s[x1 - 1][y2] - s[x2][y1 - 1] + s[x1 - 1][y1 - 1];
}
```
2. 一维差分
```cpp
int n;                      // 数组长度
int a[N];                   // 原数组（下标从1开始）
int b[N];                   // 差分数组

// 构建差分数组：b[i] = a[i] - a[i-1]
// 调用前需先读入 a[i]，并设置 n
void build_diff_1d() {
    for (int i = 1; i <= n; i ++ ) 
        b[i] = a[i] - a[i - 1];
}

// 区间加：将 a[l..r] 都加上 c
// 时间复杂度 O(1)
void add_diff_1d(int l, int r, int c) {
    b[l] += c;
    b[r + 1] -= c;      // 注意：r+1 不能越界（数组开大一点）
}

// 还原原数组（前缀和）：求每个位置最终的值
// 调用后 a[i] 即为最终值
void restore_diff_1d() {
    for (int i = 1; i <= n; i ++ ) {
        b[i] += b[i - 1];       // b 变成前缀和
        a[i] = b[i];            // 赋值回原数组（或直接使用 b[i]）
    }
}
```

3. 二维差分
```cpp
int n, m;                   // 矩阵行数、列数
int a[N][N];                // 原矩阵（下标从1开始）
int b[N][N];                // 二维差分数组

// 构建二维差分数组
// 调用前需先读入 a[i][j]，并设置 n, m
void build_diff_2d() {
    // 方法：对每个 a[i][j] 执行一次单点加
    // 等价于在 (i,j) 处加 a[i][j]，在 (i+1,j) 减，在 (i,j+1) 减，在 (i+1,j+1) 加
    for (int i = 1; i <= n; i ++ )
        for (int j = 1; j <= m; j ++ )
            add_diff_2d(i, j, i, j, a[i][j]);
}
// 子矩阵加：将 (x1,y1) 到 (x2,y2) 全部加上 c
// 时间复杂度 O(1)
void add_diff_2d(int x1, int y1, int x2, int y2, int c) {
    b[x1][y1] += c;
    b[x2 + 1][y1] -= c;
    b[x1][y2 + 1] -= c;
    b[x2 + 1][y2 + 1] += c;
}
// 还原原数组（二维前缀和）：求每个位置最终的值
// 调用后 a[i][j] 即为最终值
void restore_diff_2d() {
    for (int i = 1; i <= n; i ++ )
        for (int j = 1; j <= m; j ++ ) {
            // b[i][j] 求二维前缀和
            b[i][j] += b[i - 1][j] + b[i][j - 1] - b[i - 1][j - 1];
            a[i][j] = b[i][j];      // 赋值回原数组（或直接使用 b[i][j]）
        }
}
```
4. 离散化
```cpp
int n;                      // 原数据个数
int a[N];                   // 原数据（待离散化）
vector<int> alls;           // 存所有待离散化的值（会去重排序）

// 准备离散化：将 a[1..n] 存入 alls 并去重排序
// 调用前需先读入 a[i]，并设置 n
void prepare_discrete() {
    for (int i = 1; i <= n; i ++ ) alls.push_back(a[i]);
    sort(alls.begin(), alls.end());
    alls.erase(unique(alls.begin(), alls.end()), alls.end());
}
// 二分查找：返回 x 离散化后的映射值（从 1 开始）
// 如果 x 不存在，返回 -1
int get_discrete(int x) {
    // lower_bound 返回第一个 >= x 的迭代器
    int l = 0, r = alls.size() - 1;
    while (l < r) {
        int mid = l + r >> 1;
        if (alls[mid] >= x) r = mid;
        else l = mid + 1;
    }
    // 判断是否存在
    if (alls[l] != x) return -1;
    return l + 1;   // 映射到 1-based
}
// 如果确定 x 一定存在，可以用更简洁的写法（二分）
int get_discrete_simple(int x) {
    return lower_bound(alls.begin(), alls.end(), x) - alls.begin() + 1;
}
```
